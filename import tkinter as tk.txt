import tkinter as tk
from tkinter import ttk

def sende_befehl(wert):
    # Hier würde später dein MQTT-Code stehen, um den ESP32 zu steuern
    print(f"Sende Geschwindigkeit: {wert}%")

root = tk.Tk()
root.title("LKW Control Center")
root.geometry("400x300")

# Überschrift
label = tk.Label(root, text="LKW Steuerung", font=("Arial", 16))
label.pack(pady=10)

# Geschwindigkeits-Regler
speed_slider = tk.Scale(root, from_=0, to=100, orient="horizontal", 
                        label="Gas geben", command=sende_befehl)
speed_slider.pack(pady=20, fill="x", padx=20)

# Not-Aus Button
stop_btn = tk.Button(root, text="STOPP", bg="red", fg="white", 
                     font=("Arial", 12, "bold"), command=lambda: sende_befehl(0))
stop_btn.pack(pady=20)

root.mainloop()
